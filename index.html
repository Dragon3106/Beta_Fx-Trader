Daraja Backend Code (Node.js) - Copy to Render.com
const express=require('express');const axios=require('axios');const app=express();app.use(express.json());app.use(require('cors')());
let consumerKey='YOUR_KEY';let consumerSecret='YOUR_SECRET';let shortcode='174379';let passkey='YOUR_PASSKEY';
async function getToken(){let auth=Buffer.from(consumerKey+':'+consumerSecret).toString('base64');let r=await axios.get('https://sandbox.safaricom.co.ke/oauth/v1/generate?grant_type=client_credentials',{headers:{Authorization:'Basic '+auth}});return r.data.access_token;}
app.post('/stkpush',async(req,res)=>{try{let {phone,amount}=req.body;let token=await getToken();let timestamp=new Date().toISOString().replace(/[^0-9]/g,'').slice(0,14);let password=Buffer.from(shortcode+passkey+timestamp).toString('base64');let r=await axios.post('https://sandbox.safaricom.co.ke/mpesa/stkpush/v1/processrequest',{BusinessShortCode:shortcode,Password:password,Timestamp:timestamp,TransactionType:'CustomerPayBillOnline',Amount:amount,PartyA:phone,PartyB:shortcode,PhoneNumber:phone,CallBackURL:'https://yourdomain.com/callback',AccountReference:'BETA_FX',TransactionDesc:'Deposit'},{headers:{Authorization:'Bearer '+token}});res.json(r.data);}catch(e){res.status(500).json({error:e.message})}});
app.listen(3000,()=>console.log('Daraja backend running'));
